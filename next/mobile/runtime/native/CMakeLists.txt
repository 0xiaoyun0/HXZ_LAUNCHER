cmake_minimum_required(VERSION 3.22)
project(hxz_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hxz_runtime SHARED hxz_android.cpp hxz_jvm.cpp hxz_glfw.cpp hxz_ipc.cpp)
target_link_libraries(hxz_runtime android log dl EGL)
add_executable(hxz_jvm_exec hxz_jvm_exec.cpp)
set_target_properties(hxz_jvm_exec PROPERTIES OUTPUT_NAME libhxz_jvm_exec.so RUNTIME_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
target_link_libraries(hxz_jvm_exec hxz_runtime dl log)
foreach(target hxz_runtime hxz_jvm_exec)
  target_link_options(${target} PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endforeach()
